            if (status == EXCEPTION_SINGLE_STEP && pending_probe >= 0) {
                context.EFlags &= ~0x100;
                if (!write_bytes(child.hProcess, probe_sites[pending_probe], &trap, 1)) failed = 1;
                pending_probe = -1;
                SetThreadContext(thread, &context);
            } else if (status == EXCEPTION_BREAKPOINT) {
                if (initial_break && (address < 0x401000 || address >= 0x4104c6)) {
                    initial_break = 0;
                    if (probes_enabled) for (i = 0; i < probe_count; ++i) {
                        if (!read_bytes(child.hProcess, probe_sites[i], &probe_bytes[i], 1) ||
                            !write_bytes(child.hProcess, probe_sites[i], &trap, 1)) failed = 1;
                    }
                } else {
                    known = 0;
                    for (i = 0; probes_enabled && i < probe_count; ++i) if (address == probe_sites[i]) {
                        fprintf(log, "%s %08lx eax %08lx\n", probe_names[i], address, context.Eax);
                        state(log, child.hProcess, &context, address, 0);
                        if (!write_bytes(child.hProcess, address, &probe_bytes[i], 1)) failed = 1;
                        context.Eip = address;
                        context.EFlags |= 0x100;
                        pending_probe = i;
                        SetThreadContext(thread, &context);
                        known = 1;
                    }
                    for (i = 0; !known && i < range_count; ++i) if (ranges[i].start == address && !ranges[i].reached) {
                        count = ranges[i].end - address;
                        if (!count || count > sizeof(original_bytes) ||
                            fseek(original, ranges[i].offset, SEEK_SET) ||
                            fread(original_bytes, 1, count, original) != count ||
                            !write_bytes(child.hProcess, address, original_bytes, count)) failed = 1;
                        /* Restoring this function must also restore its probes,
                         * whose bytes were still INT3 in the initial masked image. */
                        if (probes_enabled) {
                            unsigned p;
                            for (p = 0; p < probe_count; ++p) if (probe_sites[p] >= address && probe_sites[p] < ranges[i].end) {
                                if (!read_bytes(child.hProcess, probe_sites[p], &probe_bytes[p], 1) ||
                                    !write_bytes(child.hProcess, probe_sites[p], &trap, 1)) failed = 1;
                            }
                        }
                        ranges[i].reached = 1;
                        fprintf(log, "entry %08lx\n", address);
                        state(log, child.hProcess, &context, address, 0);
                        context.Eip = address;
                        SetThreadContext(thread, &context);
                        known = 1;
                    }
