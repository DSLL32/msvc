#!/usr/bin/env python3
"""Actual four-argument directive writer boundary."""
import json
from pathlib import Path
import re
import shutil
import tempfile
import time
from compare import compare
from state import sha


def main():
    import argparse
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--original',type=Path,required=True);parser.add_argument('--native',type=Path,required=True);parser.add_argument('--output',type=Path,required=True)
    args=parser.parse_args();begin=time.monotonic();positive=compare(args.original,args.native);passed=[]
    def write(path,value):path.write_text(json.dumps(value,indent=2)+'\n')
    def native(root,section,change):
        path=root/'n/state.json';state=json.loads(path.read_text());change(state[section]);write(path,state)
        path=root/'n/gdb.stdout';prefix='C2-COFF-DIRECTIVE-OUTPUT'if section=='coff_directive_output'else'C2-OBJECT-WRITER'if section=='object_writer'else'C2-GLOBALS-COMPLETE'
        path.write_text(re.sub('^'+prefix+r' .*$',lambda m:prefix+' '+json.dumps(state[section],sort_keys=True),path.read_text(),flags=re.M))
        path=root/'n/receipt.json';receipt=json.loads(path.read_text());receipt['state_sha256']=sha((root/'n/state.json').read_bytes());receipt['gdb_outputs']['gdb.stdout']=sha((root/'n/gdb.stdout').read_bytes());write(path,receipt)
    def original(root,old,new):
        path=root/'o/events.txt';before=path.read_text();assert old in before;path.write_text(before.replace(old,new))
        path=root/'o/receipt.json';receipt=json.loads(path.read_text());receipt['events_sha256']=sha((root/'o/events.txt').read_bytes());write(path,receipt)
    def arena(root,offset):
        def change(value):
            writer=value['writer'];data=bytearray.fromhex(writer['pool_bytes']);data[offset]^=1;writer['pool_bytes']=data.hex()
        native(root,'globals_complete',change)
    controls=[('prefix formal omitted',lambda r:native(r,'coff_directive_output',lambda v:v.__setitem__('formals',v['formals'][1:]))),
        ('quoted formal omitted',lambda r:native(r,'coff_directive_output',lambda v:v.__setitem__('formals',v['formals'][:-1]))),
        ('suffix formal wrong type',lambda r:native(r,'coff_directive_output',lambda v:v['formals'][2].__setitem__('type','int'))),
        ('actual directive prefix differs',lambda r:native(r,'coff_directive_output',lambda v:v.__setitem__('prefix','-nodefaultlib:'))),
        ('actual directive value differs',lambda r:native(r,'coff_directive_output',lambda v:v.__setitem__('value','WRONG'))),
        ('actual suffix no longer NULL',lambda r:native(r,'coff_directive_output',lambda v:v.__setitem__('suffix_null',False))),
        ('actual quote mode differs',lambda r:native(r,'coff_directive_output',lambda v:v.__setitem__('quoted',1))),
        ('actual value loses caller payload alias',lambda r:native(r,'coff_directive_output',lambda v:v.__setitem__('caller_value_alias',False))),
        ('original actual value pointer differs',lambda r:original(r,'observe 00453eab eax=', 'observe 00453eab eax=0')),
        ('original suffix stack argument differs',lambda r:original(r,'observe-stack 00453e27 00000000 00000000','observe-stack 00453e27 00000001 00000000')),
        ('original quote stack argument differs',lambda r:original(r,'observe-stack 00453e27 00000000 00000000','observe-stack 00453e27 00000000 00000001'))]
    with tempfile.TemporaryDirectory(prefix='coff-directive-output-controls-',dir=Path.home()/'.ai-tmp/msvc2/vc5.10/c2')as folder:
        for index,(name,change)in enumerate(controls):
            root=Path(folder)/str(index);shutil.copytree(args.original,root/'o');shutil.copytree(args.native,root/'n');change(root)
            try:compare(root/'o',root/'n')
            except (AssertionError,KeyError,ValueError)as error:passed.append({'control':name,'reason':str(error)})
            else:raise AssertionError('coff-directive-output mutation accepted: '+name)
    args.output.write_text(json.dumps({'positive':positive,'count':len(passed),'rejected':passed,'seconds':time.monotonic()-begin,'original_receipt_sha256':sha((args.original/'receipt.json').read_bytes()),'native_receipt_sha256':sha((args.native/'receipt.json').read_bytes()),'checker_sha256':sha(Path(__file__).read_bytes())},indent=2)+'\n')

if __name__=='__main__':main()
